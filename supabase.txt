-- Create the main comments table supporting views, filters, threading, and authors
create table view_comments (
    id uuid default gen_random_uuid() primary key,
    view_hash text not null,
    author text not null,
    comment_text text not null,
    parent_id uuid references view_comments(id) on delete cascade,
    created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- Enable Row Level Security (RLS) for data safety
alter table view_comments enable row level security;

-- Create a policy allowing public read/write access for your static site users
create policy "Allow public comments" 
on view_comments 
for all 
using (true) 
with check (true);

-- ---------------------------------------------------------------------------
-- Login accounts for the comment tools.
--
-- netlify/functions/api.js authenticates against this table, so adding or
-- removing a user is an INSERT/DELETE here — no Netlify redeploy needed.
-- ---------------------------------------------------------------------------
create table app_users (
    id uuid default gen_random_uuid() primary key,
    username text unique not null,
    password text not null,
    created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- Enable Row Level Security (RLS)
alter table app_users enable row level security;

-- Public read access: the Netlify function reads the table with the anon key.
-- NOTE: this also means anyone holding that (public) key can read the rows, so
-- either store hashes in `password` (`npm run add-user -- <user> <pw> --hash`)
-- or set SUPABASE_SERVICE_ROLE_KEY in Netlify and replace this policy with a
-- private one. Read-only: anonymous clients can never insert or update.
create policy "Allow public login verification"
on app_users
for select
using (true);

-- First account (password may be plain text or a `salt:sha256hex` hash).
